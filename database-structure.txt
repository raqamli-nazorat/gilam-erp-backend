Table Country {
  id   integer [primary key] // uuid 16
  name varchar
  is_active  boolean [default: true]
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table Region {
  id   integer [primary key] // uuid 16
  name varchar
  country integer [ref: > Country.id, not null]
  is_active  boolean [default: true]
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table District {
  id         integer [primary key] // uuid 16
  region     integer [ref: > Region.id, not null]
  name       varchar
  is_active  boolean [default: true]
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table Organization {
  id         integer [primary key] // uuid 16
  name       varchar(255) [not null]
  inn        varchar(20) [unique, not null]
  phone      varchar(50)
  director   varchar(50)
  region     integer [ref: > Region.id, not null]
  district   integer [ref: > District.id, not null]
  address    text
  prefix     varchar
  is_active  boolean [default: true]
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table Branch {
  id              integer [primary key] // uuid 16
  organization    integer [ref: > Organization.id, not null]
  name            varchar(255) [not null]
  phone           varchar(50)
  region          integer [ref: > Region.id, not null]
  district        integer [ref: > District.id, not null]
  address         text
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table Role {
  id              integer [primary key] // uuid 16
  name            varchar(255) [not null]
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table User {
  id              integer [primary key]      // uuid 16
  branch_id       integer [ref: > Branch.id, not null]
  branch          integer [ref: > Branch.id,  not null] // NULL bo'lsa - bosh ofis yoki global admin
  full_name       varchar(255) [not null]    // username
  phone_number    varchar(50)
  role            integer [ref: > Role.id, not null]
  password_hash   varchar(255) [not null]
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table Employee {
  full_name       varchar
  region          integer [ref: >? Region.id, null]
  district        integer [ref: >? District.id, null]
  address         varchar
  passport_seria  varchar
  passport_number varchar
  jsshr           varchar [null]
  stir            varchar [null]
  phone_number    varchar [null]
  id              integer [primary key, increment]
  branch_id       integer [ref: > Branch.id, not null]
  name            varchar(100) [not null]
  description     text
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table Position {
  id           integer [primary key, increment]
  name         varchar(100) [not null]
  description  text
  is_active    boolean [default: true]
  created_at   timestamp [default: `now()`]
  updated_at   timestamp [default: `now()`]
}

Table WorkSchedule {
  id           integer [primary key, increment]
  branch_id    integer [ref: > Branch.id, not null]
  name         varchar(100) [not null]
  description  text
  from_date    date [not null]
  to_date      date [not null]
  from_hour    hour [not null]
  to_hour      hour [not null]
  is_active    boolean [default: true]
  created_at   timestamp [default: `now()`]
  updated_at   timestamp [default: `now()`]
}

Table WorkScheduleItem {
  id           integer [primary key, increment]
  workSchedule integer [ref: > Organization.id, not null]
  name         varchar(100) [not null]
  day_type     choice("To'liq bayram", "Qisman ish kuni", "To'liq ish kuni")
  day_date         date [not null]
  from_hour    hour [not null]
  to_hour      hour [not null]
  is_active    boolean [default: true]
  created_at   timestamp [default: `now()`]
  updated_at   timestamp [default: `now()`]
}

Table RecruitmentDismissal {
  type             choice ("Ishga olish | recruitment", "Ishdan chiqarish | dismissal")
  branch_id        integer [ref: > Branch.id, not null]
  employee         integer [ref: > Employee.id, not null]
  position         integer [ref: > Position.id, not null]
  branch           integer [ref: > Branch.id, not null]
  card_number      varchar [not null]
  card_image       varchar // bejik rasmi
  salary_type      choice ("Belgilangan summa", "Savdodan foiz", "Asoschi") [not null]
  fix_summa        float     [null]
  fix_percent      float     [null]
  rec_dism_date    date      [not null]
  dismissal_reason text
  extra_summa      float     [null]
  extra_percent    float     [null]
  id               integer   [primary key, increment]
  is_active        boolean   [default: true]
  created_at       timestamp [default: `now()`]
  updated_at       timestamp [default: `now()`]
}

Table EmployeeLedger {
  id              integer [primary key]      // uuid 16
  branch_id       integer [ref: > Branch.id, not null]
  employee        integer [ref: > Employee.id, not null]
  type            choice  ("recruitment", "dismissal_work", "change_position")
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]

}
Table Currency {
  id              integer [primary key]      // uuid 16
  name            varchar // O'zbekiston so'mi
  short_name      varchar // UZS
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table CurrencyLedger { // o'zbek so'miga nisbatan qiymatlar saqlanadi.
  id              integer [primary key]      // uuid 16
  currency        integer [ref: > Currency.id, not null]
  value           float   [default: 0.00]
  day             timestamp [default: `now()`]
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

Table AccrualRetention {
  id              integer [primary key]      // uuid 16
  name            varchar
  type            choice ("percent", "fix_summa")
  currency        integer [ref: > Currency.id, not null]
  value           float   [default: 0.00] // if type == "percent": max(value) = 100.00 else min(value) = 0 | Hisob kitobda agar boshqa valyuta tanlangan bo'lsa, bu maydon qimayti oylik hisoblash paytida UZS ga joriy sanadagi kurs bilan almashtiriladi, ya'ni oylik UZSdan xisoblanadi.
  is_active       boolean [default: true]
  created_at      timestamp [default: `now()`]
  updated_at      timestamp [default: `now()`]
}

  id           integer [primary key, increment]
  branch_id    integer [ref: > Branch.id, not null]
